    <script>
        // 設定點擊格子時要切換的狀態與顏色
        const states = [
            { text: '', color: 'inherit' },
            { text: '✗', color: '#e74c3c' }, // 紅色代表沒有
            { text: '◯', color: '#2ecc71' }, // 綠色代表有
            { text: '？', color: '#3498db' }  // 藍色代表疑問
        ];

        document.querySelectorAll('.check-cell').forEach(cell => {
            // 初始化狀態索引
            cell.dataset.stateIndex = 0;

            cell.addEventListener('click', function() {
                // 計算下一個狀態
                let currentIndex = parseInt(this.dataset.stateIndex);
                let nextIndex = (currentIndex + 1) % states.length;
                
                // 更新畫面與資料
                this.dataset.stateIndex = nextIndex;
                this.innerText = states[nextIndex].text;
                this.style.color = states[nextIndex].color;
            });
        });
    </script>
